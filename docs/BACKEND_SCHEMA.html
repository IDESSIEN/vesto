1. Data Architecture Principles

Smart contracts handle money. Supabase handles metadata. These two concerns must never bleed into each other. The contract is the source of truth for USDC balances and invoice funding states. Supabase is the source of truth for identity, documents, descriptions, and admin notes. Any sync between them flows one way: onchain events write to Supabase via Edge Functions. Supabase never writes to the contract.

2. Core Tables

2.1 profiles

CREATE TABLE profiles ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE UNIQUE, role TEXT NOT NULL CHECK (role IN ('seller', 'lender', 'admin')), full_name TEXT NOT NULL, email TEXT, wallet_address TEXT, -- connected EVM wallet created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() );

2.2 seller_profiles

CREATE TABLE seller_profiles ( id UUID PRIMARY KEY REFERENCES profiles(id) ON DELETE CASCADE, business_name TEXT NOT NULL, operating_country TEXT NOT NULL, category TEXT NOT NULL, phone_prefix TEXT, phone_number TEXT, tax_id TEXT, verification_tier SMALLINT NOT NULL DEFAULT 0, -- 0, 1, 2 kyc_status_tier1 TEXT DEFAULT 'none', -- none|pending|verified|rejected kyc_status_tier2 TEXT DEFAULT 'none', credit_limit NUMERIC(12,2) NOT NULL DEFAULT 0, used_limit NUMERIC(12,2) NOT NULL DEFAULT 0, total_financed NUMERIC(12,2) NOT NULL DEFAULT 0, tier1_doc_url TEXT, tier2_doc_url TEXT, tier2_video_url TEXT, bank_statement_url TEXT );

2.3 lender_profiles

CREATE TABLE lender_profiles ( id UUID PRIMARY KEY REFERENCES profiles(id) ON DELETE CASCADE, account_type TEXT NOT NULL DEFAULT 'individual', investor_tier TEXT DEFAULT 'retail', target_allocation NUMERIC(12,2) DEFAULT 0, total_invested NUMERIC(12,2) DEFAULT 0, total_yield_earned NUMERIC(12,2) DEFAULT 0, risk_accepted BOOLEAN DEFAULT FALSE, risk_accepted_at TIMESTAMPTZ, auto_invest_enabled BOOLEAN DEFAULT FALSE );

2.4 invoices

CREATE TABLE invoices ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), seller_id UUID NOT NULL REFERENCES seller_profiles(id), contract_invoice_id BYTEA, -- bytes32 hash used onchain buyer_name TEXT NOT NULL, buyer_tax_id TEXT, buyer_country TEXT, amount NUMERIC(12,2) NOT NULL, advance_rate_pct SMALLINT NOT NULL, -- 85 or 90 advance_amount NUMERIC(12,2) NOT NULL, fee_pct NUMERIC(5,2) NOT NULL DEFAULT 2.0, fee_amount NUMERIC(12,2) NOT NULL, expected_yield_pct NUMERIC(5,2), due_date DATE NOT NULL, term_days SMALLINT, risk_tier TEXT, -- A+, A, B+, B risk_score SMALLINT, -- 0-100 status TEXT NOT NULL DEFAULT 'pending_admin_approval', doc_name TEXT, doc_url TEXT, funded_by_lender_id UUID REFERENCES lender_profiles(id), funded_at TIMESTAMPTZ, repaid_at TIMESTAMPTZ, flag_reason TEXT, seller_verification_tier SMALLINT, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); CREATE INDEX idx_invoices_seller_id ON invoices(seller_id); CREATE INDEX idx_invoices_status ON invoices(status); CREATE INDEX idx_invoices_funded_by ON invoices(funded_by_lender_id);

2.5 verification_requests

CREATE TABLE verification_requests ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), seller_id UUID NOT NULL REFERENCES seller_profiles(id), tier SMALLINT NOT NULL CHECK (tier IN (1, 2)), document_type TEXT NOT NULL, document_url TEXT, video_url TEXT, status TEXT NOT NULL DEFAULT 'pending', risk_score_suggested SMALLINT, cleanverse_status TEXT, -- pass | fail | uncertain cleanverse_confidence NUMERIC(5,2), cleanverse_reason TEXT, submitted_at TIMESTAMPTZ DEFAULT NOW(), reviewed_at TIMESTAMPTZ, reviewed_by UUID REFERENCES profiles(id) );

2.6 admin_notes

CREATE TABLE admin_notes ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), admin_id UUID NOT NULL REFERENCES profiles(id), entity_type TEXT NOT NULL, -- 'invoice' | 'verification' | 'dispute' entity_id UUID NOT NULL, note TEXT NOT NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE INDEX idx_admin_notes_entity ON admin_notes(entity_type, entity_id);

2.7 disputes

CREATE TABLE disputes ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), invoice_id UUID NOT NULL REFERENCES invoices(id), raised_by_id UUID NOT NULL REFERENCES profiles(id), raised_by_role TEXT NOT NULL, reason TEXT NOT NULL, status TEXT NOT NULL DEFAULT 'open', -- open | resolved resolution TEXT, -- refund_lender | pay_seller resolved_by_id UUID REFERENCES profiles(id), created_at TIMESTAMPTZ DEFAULT NOW(), resolved_at TIMESTAMPTZ );

3. Row-Level Security Policies

-- Enable RLS on all tables ALTER TABLE profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE seller_profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE lender_profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE invoices ENABLE ROW LEVEL SECURITY; ALTER TABLE verification_requests ENABLE ROW LEVEL SECURITY; ALTER TABLE admin_notes ENABLE ROW LEVEL SECURITY; ALTER TABLE disputes ENABLE ROW LEVEL SECURITY; -- ── SELLER POLICIES ────────────────────────────────────────────── -- Sellers can only see and edit their own profile CREATE POLICY seller_own_profile ON seller_profiles FOR ALL USING (id = ( SELECT id FROM profiles WHERE user_id = auth.uid() )); -- Sellers can only see their own invoices (+ published marketplace ones) CREATE POLICY seller_own_invoices ON invoices FOR SELECT USING ( seller_id = (SELECT id FROM profiles WHERE user_id = auth.uid()) OR status = 'published_marketplace' ); CREATE POLICY seller_insert_invoices ON invoices FOR INSERT WITH CHECK ( seller_id = (SELECT id FROM profiles WHERE user_id = auth.uid()) ); -- ── LENDER POLICIES ────────────────────────────────────────────── -- Lenders can see their own profile + published invoices CREATE POLICY lender_marketplace ON invoices FOR SELECT USING (status = 'published_marketplace'); CREATE POLICY lender_own_portfolio ON invoices FOR SELECT USING ( funded_by_lender_id = (SELECT id FROM lender_profiles WHERE id = ( SELECT id FROM profiles WHERE user_id = auth.uid() )) ); -- ── ADMIN POLICIES ─────────────────────────────────────────────── -- Admins have full access to all tables CREATE POLICY admin_all_access ON invoices FOR ALL USING ( EXISTS (SELECT 1 FROM profiles WHERE user_id = auth.uid() AND role = 'admin') ); -- (Same pattern repeated for all tables)

4. Supabase Storage Buckets

BucketPath PatternAccessContents
kyc-documents{seller_id}/tier{1|2}/{filename}Seller: own path. Admin: all paths. Lender: none.National ID photos, passport scans, tax PIN PDFs, bank statements
invoice-documents{seller_id}/{invoice_id}/{filename}Seller: own path. Lender: read published. Admin: all.Commercial invoice PDFs, bills of lading, purchase orders
verification-videos{seller_id}/tier2/selfie_{timestamp}.webmSeller: own path. Admin: all. Lender: none.Tier 2 video selfie recordings

Retention policy: KYC documents are retained for the duration of the seller's active account. On account deletion, documents are purged within 30 days per GDPR Article 17. Purge is triggered by a Supabase Edge Function called by the account deletion webhook.

5. Edge Function Contracts

5.1 process-verification

Triggered by: admin action (approve/reject in VerificationQueue). Updates verification_requests.status, writes to admin_notes, and updates seller_profiles.verification_tier and credit_limit atomically.

// Input { verification_request_id: string, action: 'approve' | 'reject', admin_note?: string } // On approve: // 1. Set verification_requests.status = 'approved' // 2. Set seller_profiles.verification_tier = request.tier // 3. Set seller_profiles.credit_limit = tier === 1 ? 500 : 5000 // 4. Set seller_profiles.kyc_status_tier{n} = 'verified' // 5. Insert admin_notes record // 6. Send notification to seller (email / webhook) // On reject: // 1. Set verification_requests.status = 'rejected' // 2. Set seller_profiles.kyc_status_tier{n} = 'rejected' // 3. Insert admin_notes with rejection reason

5.2 sync-onchain-event

Triggered by: Arc RPC event listener (polling InvoiceFunded, PayoutClaimed, DisputeResolved events). Keeps Supabase invoice status in sync with the contract state.

// Input: decoded Arc event { event: 'InvoiceFunded' | 'PayoutClaimed' | 'DisputeResolved', args: any, txHash: string } // InvoiceFunded → invoices.status = 'funded', invoices.funded_at = now() // PayoutClaimed → invoices.status = 'repaid', invoices.repaid_at = now() // DisputeResolved → invoices.status = 'repaid' | 'flagged'

5.3 aggregate-analytics

Triggered by: cron job every 15 minutes. Computes platform-level metrics and writes to platform_analytics materialised view. Served to the admin Analytics Overview screen.

// Output written to platform_analytics table { total_volume_usd: number, // sum(invoices.amount) where status != 'pending' active_liquidity_usd: number, // sum(invoices.advance_amount) where status = 'funded' average_yield_apy: number, // avg(invoices.expected_yield_pct) where status = 'funded' default_rate: number, // count(disputed) / count(funded) invoices_funded_count: number, avg_advance_rate: number, computed_at: string // ISO timestamp }

6. Onchain ↔ Off-chain Sync Pattern

// Arc event listener (server-side, bun runtime) // Polls logs every 12 seconds (Arc block time ~1s; 12s gives safe backlog) const rpcUrl = process.env.RPC_PROXY_BASE_URL + '/api/rpc/Arc_Testnet?_rpc_token=' + process.env.RPC_PROXY_TOKEN; const client = createPublicClient({ transport: http(rpcUrl) }); client.watchContractEvent({ address: VESTO_ESCROW_ADDRESS, abi: vestoEscrowAbi, onLogs: async (logs) => { for (const log of logs) { await fetch(`${SUPABASE_URL}/functions/v1/sync-onchain-event`, { method: 'POST', headers: { Authorization: `Bearer ${SUPABASE_SERVICE_KEY}` }, body: JSON.stringify({ event: log.eventName, args: log.args, txHash: log.transactionHash }) }); } } });

7. Schema Evolution — V2 & V3

V2 Additions (3–9 months)

-- V2: lender pools (pooled capital allocation) CREATE TABLE lender_pools ( id UUID PRIMARY KEY, name TEXT NOT NULL, total_capital NUMERIC(18,6), available_capital NUMERIC(18,6), risk_strategy TEXT, -- conservative | balanced | aggressive min_risk_score SMALLINT, max_term_days SMALLINT, created_at TIMESTAMPTZ DEFAULT NOW() ); -- V2: yield accrual ledger CREATE TABLE yield_accrual_ledger ( id UUID PRIMARY KEY, lender_id UUID REFERENCES lender_profiles(id), invoice_id UUID REFERENCES invoices(id), gross_pct NUMERIC(5,2), fee_pct NUMERIC(5,2), net_pct NUMERIC(5,2), usdc_earned NUMERIC(12,6), accrued_at TIMESTAMPTZ DEFAULT NOW() ); -- V2: multi-chain invoice mapping (CCTP) CREATE TABLE multi_chain_invoice_map ( invoice_id UUID REFERENCES invoices(id), chain_id BIGINT NOT NULL, contract_address TEXT NOT NULL, bridge_tx_hash TEXT, status TEXT DEFAULT 'pending', PRIMARY KEY (invoice_id, chain_id) );
